/// PREPARATION ONLY. Not wired into any replay before the complete Kit raw gate is available.
private enum FoundationPayloadProbe {
    static func visit(_ value: Any) throws -> ScenarioPayload {
        if value is NSNull { return .scalar(.null) }
        if let object = value as? [String: Any] { return .object(try object.mapValues(visit)) }
        if let array = value as? [Any] { return .array(try array.map(visit)) }
        if let text = value as? String { return .scalar(.text(text)) }
        if let number = value as? NSNumber {
            if CFGetTypeID(number) == CFBooleanGetTypeID() { return .scalar(.flag(number.boolValue)) }
            let value = number.doubleValue
            guard value.isFinite else { throw ConformanceError.nonFiniteNumber }
            return .scalar(.number(value))
        }
        throw ConformanceError.invalidScalar
    }

    static func decode(_ data: Data) throws -> ScenarioPayload {
        try visit(JSONSerialization.jsonObject(with: data, options: [.fragmentsAllowed]))
    }

    static func encoded(_ value: ScenarioPayload) throws -> Data {
        let encoder = JSONEncoder(); encoder.outputFormatting = [.sortedKeys]
        return try encoder.encode(value)
    }

    static func exactDifferences(
        _ left: ScenarioPayload,
        _ right: ScenarioPayload,
        path: String = "$",
        result: inout [[String: String]]
    ) {
        switch (left, right) {
        case let (.scalar(.number(a)), .scalar(.number(b))):
            if a.bitPattern != b.bitPattern { result.append([
                "path": path,
                "kind": "Double.bitPattern",
                "old": String(a.bitPattern, radix: 16),
                "foundation": String(b.bitPattern, radix: 16)
            ]) }
        case let (.scalar(.text(a)), .scalar(.text(b))):
            if Array(a.utf8) != Array(b.utf8) { result.append([
                "path": path,
                "kind": "String.UTF8",
                "old": Data(a.utf8).base64EncodedString(),
                "foundation": Data(b.utf8).base64EncodedString()
            ]) }
        case let (.scalar(.flag(a)), .scalar(.flag(b))):
            if a != b { result.append(["path": path, "kind": "Bool", "old": String(a), "foundation": String(b)]) }
        case (.scalar(.null), .scalar(.null)): break
        case let (.array(a), .array(b)):
            guard a.count == b.count else { result.append(["path": path, "kind": "array.count"]); return }
            for i in a.indices { exactDifferences(a[i], b[i], path: path + "[" + String(i) + "]", result: &result) }
        case let (.object(a), .object(b)):
            let ak = a.keys.map { Data($0.utf8).base64EncodedString() }.sorted()
            let bk = b.keys.map { Data($0.utf8).base64EncodedString() }.sorted()
            guard ak == bk else { result.append(["path": path, "kind": "object.key.UTF8"]); return }
            for key in a.keys { exactDifferences(a[key]!, b[key]!, path: path + "." + key, result: &result) }
        default: result.append([
                "path": path,
                "kind": "JSON scalar/container type",
                "old": String(describing: left),
                "foundation": String(describing: right)
            ])
        }
    }
}

final class J4FoundationPayloadProbeTests: XCTestCase {
    func testCurrent49adPayloadAndFoundationAcceptanceAndBitFidelity() throws {
        let textCases: [(String, String)] = [
            ("scalar_types", #"{"v":[true,false,1,0,-1,1.25,null,"18446744073709551615","unknown",{"x":null}]}"#),
            ("nfc", "{\"v\":\"\u{e9}\"}"), ("nfd", "{\"v\":\"e\u{301}\"}"),
            ("utf8", "{\"v\":\"控制台🚀\"}"), ("surrogate_pair", #"{"v":"\uD83D\uDE80"}"#),
            ("escaped_nul", #"{"v":"\u0000"}"#), ("escaped_slash_quote", #"{"v":"a\/b\"c\\d"}"#),
            ("negative_zero_integer", #"{"v":-0}"#), ("negative_zero_decimal", #"{"v":-0.0}"#),
            ("negative_zero_exponent", #"{"v":-0e0}"#), ("negative_zero_large_exponent", #"{"v":-0E-100}"#),
            ("positive_zero_decimal", #"{"v":0.0}"#), ("positive_underflow", #"{"v":0e-4000}"#),
            ("negative_underflow", #"{"v":-0e-4000}"#), ("subnormal_short", #"{"v":5e-324}"#),
            ("subnormal_exact", #"{"v":4.9406564584124654e-324}"#), ("negative_subnormal", #"{"v":-5e-324}"#),
            ("subnormal_half", #"{"v":2.4703282292062327e-324}"#), ("underflow_nonzero", #"{"v":1e-400}"#),
            ("maximum_finite", #"{"v":1.7976931348623157e308}"#), ("overflow", #"{"v":1e309}"#),
            ("large_integer", #"{"v":9007199254740993}"#), ("uint64_numeric", #"{"v":18446744073709551615}"#),
            ("uint64_string", #"{"v":"18446744073709551615"}"#),
            ("long_decimal", #"{"v":0.1000000000000000055511151231257827021181583404541015625}"#),
            ("nan", #"{"v":NaN}"#), ("infinity", #"{"v":Infinity}"#), ("trailing_comma", #"{"v":1,}"#),
            ("duplicate_key", #"{"v":1,"v":2}"#), ("canonical_equivalent_keys", "{\"\u{e9}\":1,\"e\u{301}\":2}"),
            ("giant_integer", "{\"v\":1" + String(repeating: "0", count: 400) + "}")
        ]
        var cases = textCases.map { ($0.0, Data($0.1.utf8)) }
        cases.append(("invalid_utf8", Data([123, 34, 118, 34, 58, 34, 255, 34, 125])))
        for coefficient in ["0", "-0", "1", "-1", "2.4703282292062327", "4.9406564584124654", "9.999999999999999"] {
            for exponent in [-4000, -400, -330, -325, -324, -323, -320, -310, -309, -308, 0, 308, 309, 400, 4000] {
                let token = coefficient + "e" + String(exponent)
                cases.append(("matrix:" + token, Data(("{\"v\":" + token + "}").utf8)))
            }
        }
        if let path = ProcessInfo.processInfo.environment["BCR_J4_NUMERIC_TOKEN_FILE"] {
            let document = try JSONSerialization
                .jsonObject(with: Data(contentsOf: URL(fileURLWithPath: path))) as! [String: Any]
            for token in document["tokens"] as! [String] {
                cases.append(("ci-token:" + token, Data(("{\"v\":" + token + "}").utf8)))
            }
        }
        var bits: UInt64 = 0x1d34_794e_a6b2_08c5
        for index in 0 ..< 1024 {
            bits = bits &* 6364136223846793005 &+ 1442695040888963407
            let value = Double(bitPattern: bits)
            if value.isFinite {
                let raw = try FoundationPayloadProbe.encoded(.object(["v": .scalar(.number(value))]))
                cases.append(("encoded-double-bits:" + String(index) + ":" + String(bits, radix: 16), raw))
            }
        }
        var reports: [[String: Any]] = []
        for (name, data) in cases {
            var row: [String: Any] = ["case": name, "raw_base64": data.base64EncodedString(), "raw_bytes": data.count]
            let old: ScenarioPayload?, foundation: ScenarioPayload?
            do { old = try JSONDecoder().decode(ScenarioPayload.self, from: data); row["old_accepted"] = true }
            catch { old = nil; row["old_accepted"] = false; row["old_error"] = String(describing: error) }
            do { foundation = try FoundationPayloadProbe.decode(data); row["foundation_accepted"] = true }
            catch {
                foundation = nil; row["foundation_accepted"] = false; row["foundation_error"] =
                    String(describing: error)
            }
            if let old, let foundation {
                var differences: [[String: String]] = []
                FoundationPayloadProbe.exactDifferences(old, foundation, result: &differences)
                let a = try FoundationPayloadProbe.encoded(old), b = try FoundationPayloadProbe.encoded(foundation)
                row["exact_scalar_differences"] = differences; row["encoded_bytes_equal"] = a == b
                row["old_encoded_base64"] = a.base64EncodedString(); row["foundation_encoded_base64"] = b
                    .base64EncodedString()
                row["typed_equality_only"] = old == foundation
            }
            reports.append(row)
        }
        // The probe records Foundation mismatches as evidence; it does not waive them or claim adoption.
        let output = ProcessInfo.processInfo.environment["BCR_J4_FOUNDATION_PROBE_REPORT"]
            .map { URL(fileURLWithPath: $0) }
        if let output {
            let report: [String: Any] = [
                "kit_revision": "49ad4a7b8aa1160b9369733f0c8681647a41cd51",
                "new_path_adopted": false,
                "raw_guard": "not yet shared new API; pure payload acceptance/representation probe only",
                "cases": reports
            ]
            try JSONSerialization.data(withJSONObject: report, options: [.sortedKeys, .prettyPrinted]).write(to: output)
        }
        let types = reports.first { $0["case"] as? String == "scalar_types" }!
        XCTAssertEqual(types["encoded_bytes_equal"] as? Bool, true)
        XCTAssertEqual(types["exact_scalar_differences"] as? [[String: String]], [])
        for name in ["nfc", "nfd", "utf8", "surrogate_pair", "escaped_nul", "escaped_slash_quote", "uint64_string"] {
            let value = reports.first { $0["case"] as? String == name }!
            XCTAssertEqual(value["encoded_bytes_equal"] as? Bool, true, name)
            XCTAssertEqual(value["exact_scalar_differences"] as? [[String: String]], [], name)
        }
    }
}

private enum FoundationNarrowFallbackProbe {
    /// Only called on syntactically validated JSON. Strings/escapes are skipped, never regex-replaced.
    static func reasons(_ data: Data) -> [String] {
        let bytes = Array(data)
        var index = 0
        var reasons: [String] = []
        while index < bytes.count {
            if bytes[index] == 34 {
                index += 1
                while index < bytes.count {
                    if bytes[index] == 92 { index += 2 }
                    else if bytes[index] == 34 { index += 1; break }
                    else { index += 1 }
                }
            } else if bytes[index] == 45 || (48 ... 57).contains(bytes[index]) {
                let start = index
                while index < bytes
                    .count &&
                    (bytes[index] == 45 || bytes[index] == 43 || bytes[index] == 46 || bytes[index] == 101 ||
                        bytes[index] == 69 || (48 ... 57).contains(bytes[index])) { index += 1 }
                let token = Array(bytes[start ..< index])
                if token == [45, 48] { reasons.append("negative-integer-zero") }
                if let e = token.firstIndex(where: { $0 == 101 || $0 == 69 }) {
                    var exponent = Array(token[(e + 1)...])
                    if exponent.first == 45 || exponent.first == 43 { exponent.removeFirst() }
                    while exponent.first == 48 { exponent.removeFirst() }
                    if exponent
                        .count > 3 ||
                        (exponent.count == 3 && exponent.lexicographicallyPrecedes([51, 48, 57]) == false)
                    {
                        reasons.append("explicit-exponent-magnitude-over-308")
                    }
                }
            } else { index += 1 }
        }
        return reasons
    }
}

extension J4FoundationPayloadProbeTests {
    func testProposedLexicalFallbackAgainstOriginal49adDataAPI() throws {
        let contexts: [(String, String)] = [
            ("-0", #"{"v":-0}"#), ("-0.0", #"{"v":-0.0}"#), ("-0e0", #"{"v":-0e0}"#),
            ("0e-4000", #"{"v":0e-4000}"#), ("-0e-4000", #"{"v":-0e-4000}"#),
            ("half-subnormal", #"{"v":2.4703282292062327e-324}"#), ("1e-400", #"{"v":1e-400}"#),
            ("5e-324", #"{"v":5e-324}"#), ("1e-309", #"{"v":1e-309}"#),
            ("overflow", #"{"v":1e309}"#), ("numeric-string", #"{"v":"-0 0e-4000 1e309"}"#),
            ("trailing-comma", #"{"v":1,}"#), ("duplicate-key", #"{"v":1,"v":2}"#),
            ("canonical-equivalent-key", "{\"\u{e9}\":1,\"e\u{301}\":2}")
        ]
        var reports: [[String: Any]] = []
        for (name, context) in contexts {
            let data =
                Data(
                    (#"{"schema_version":1,"owner":"probe","ordinal":"1","operation":"initialize","arguments":{"rate":{"kind":"unlimited"}},"context":"# +
                        context + "}").utf8
                )
            let reasons = FoundationNarrowFallbackProbe.reasons(data)
            var row: [String: Any] = [
                "case": name,
                "raw_base64": data.base64EncodedString(),
                "numeric_fallback_reasons": reasons
            ]
            var original: Data?
            do {
                original = try SchedulerTraceCodec
                    .encodeInput(SchedulerTraceCodec.decodeInput(data)); row["old_accepted"] = true
            } catch { row["old_accepted"] = false; row["old_error"] = String(describing: error) }
            var bare: Data?
            do {
                let value = try FoundationPayloadProbe.decode(data)
                bare = try SchedulerTraceCodec
                    .encodeInput(SchedulerTraceCodec.decodeInput(FoundationPayloadProbe.encoded(value)))
                row["bare_foundation_accepted"] = true
            } catch { row["bare_foundation_accepted"] = false; row["bare_foundation_error"] = String(describing: error)
            }
            row["bare_foundation_encoded_equal"] = original != nil && bare != nil ? original == bare : false
            if !["trailing-comma", "duplicate-key", "canonical-equivalent-key"].contains(name) {
                // Predetermined numeric selector, never catch(error)->old-success fallback.
                var selected: Data?
                do {
                    selected = reasons.isEmpty ? bare : try SchedulerTraceCodec
                        .encodeInput(SchedulerTraceCodec.decodeInput(data))
                    row["selected_accepted"] = selected != nil
                } catch { row["selected_accepted"] = false; row["selected_error"] = String(describing: error) }
                XCTAssertEqual(selected != nil, original != nil, name)
                if let original, let selected { XCTAssertEqual(selected, original, name) }
            } else {
                row["requires_complete_raw_gate_before_AST"] = true
                XCTAssertNil(original, name)
            }
            if name == "numeric-string" { XCTAssertTrue(reasons.isEmpty) }
            reports.append(row)
        }
        if let root = ProcessInfo.processInfo.environment["BCR_J4_FOUNDATION_PROBE_REPORT"] {
            let output = URL(fileURLWithPath: root).deletingPathExtension().appendingPathExtension("codec.json")
            try JSONSerialization.data(
                withJSONObject: ["kit_revision": "49ad4a7b8aa1160b9369733f0c8681647a41cd51",
                                 "new_gate_not_available": true, "not_adopted": true, "cases": reports],
                options: [.prettyPrinted, .sortedKeys]
            ).write(to: output)
        }
    }
}
